def records(value):
    """Transforme une éventuelle réponse None en liste exploitable."""
    return [item for item in (value or []) if isinstance(item, dict)]


def as_ids(value):
    """IDs uniformisés, qu'ils soient déjà des chaînes ou des entiers."""
    if value is None:
        return []
    if not isinstance(value, (list, tuple, set)):
        value = [value]
    return [str(x) for x in value if x is not None and str(x).strip()]


def value_or_dash(value):
    if value is None or value == "":
        return "—"
    if isinstance(value, (list, tuple, set)):
        return ", ".join(map(str, value)) or "—"
    if isinstance(value, dict):
        return json.dumps(value, ensure_ascii=False, default=str)
    return str(value)


def normalize_status(value):
    status = str(value or "").strip().upper().replace("-", "_").replace(" ", "_")
    return {
        "PASS": "PASSED",
        "SUCCESS": "PASSED",
        "OK": "PASSED",
        "FAIL": "FAILED",
        "ERROR": "FAILED",
        "NOTRUN": "NOT_RUN",
        "NOT_EXECUTED": "NOT_RUN",
        "PENDING": "NOT_RUN",
    }.get(status, status or "UNKNOWN")


# Couleurs volontairement assez contrastées en modes clair et sombre.
STATUS_STYLE = {
    "PASSED": ("#DCFCE7", "#166534"),
    "FAILED": ("#FEE2E2", "#991B1B"),
    "ASSERT": ("#F3E8FF", "#6B21A8"),
    "SKIPPED": ("#E2E8F0", "#334155"),
    "NOT_RUN": ("#E2E8F0", "#334155"),
    "BLOCKED": ("#FEF3C7", "#92400E"),
    "UNKNOWN": ("#E2E8F0", "#334155"),
}


def show_status(value):
    """Texte du tableau : le point coloré reste visible sans couleur CSS."""
    return f"● {normalize_status(value)}"


def status_css(value):
    status = normalize_status(str(value).removeprefix("● "))
    background, foreground = STATUS_STYLE.get(status, STATUS_STYLE["UNKNOWN"])
    return (
        f"background-color: {background}; color: {foreground}; "
        "font-weight: 700;"
    )


def style_status(df, column="Statut"):
    """Style compatible avec la sélection de ligne de st.dataframe."""
    if column not in df.columns or df.empty:
        return df
    return df.style.apply(
        lambda serie: [status_css(value) for value in serie],
        subset=[column],
    )


def status_badge(value):
    status = normalize_status(value)
    colors = {
        "PASSED": "green",
        "FAILED": "red",
        "ASSERT": "violet",
        "BLOCKED": "orange",
        "SKIPPED": "gray",
        "NOT_RUN": "gray",
    }
    # Seules des valeurs autorisées sont interpolées dans le Markdown.
    if status not in colors:
        return ":gray-badge[INCONNU]"
    return f":{colors[status]}-badge[{status}]"


def stable_key(prefix, *parts):
    """Clé différente par campagne/filtre ; évite une ancienne sélection d'index."""
    signature = "|".join(str(p) for p in parts)
    digest = hashlib.sha1(signature.encode("utf-8")).hexdigest()[:12]
    return f"{prefix}_{digest}"

def as_table(df, status_column="Statut", *, key=None, height=None, selectable=False):
    """Point d'entrée commun : toutes les tables ont le même rendu."""
    kwargs = {
        "use_container_width": True,
        "hide_index": True,
    }
    if height is not None:
        kwargs["height"] = height
    if selectable:
        kwargs.update({
            "key": key,
            "on_select": "rerun",
            "selection_mode": "single-row",
        })
    return st.dataframe(style_status(df, status_column), **kwargs)


def get_filtered_data(campaign_id, campaigns, requirements, defects, bridge):
    if campaign_id is None:
        return [], [], []

    suites = records(bridge.get_test_suites(campaign_id=campaign_id))
    suite_ids = {str(s.get("id")) for s in suites}

    # Fonctionne même si la relation n'existe que d'un côté (suite -> exigence
    # ou exigence -> suite).
    req_ids = {rid for s in suites for rid in as_ids(s.get("requirements_ids"))}
    selected_reqs = [
        r for r in requirements
        if str(r.get("id")) in req_ids
        or bool(set(as_ids(r.get("suites_ids"))) & suite_ids)
    ]
    selected_req_ids = {str(r.get("id")) for r in selected_reqs}

    defect_ids = {did for s in suites for did in as_ids(s.get("defects_ids"))}
    defect_ids.update(
        did for r in selected_reqs for did in as_ids(r.get("defects_ids"))
    )
    selected_defects = [
        d for d in defects
        if str(d.get("id")) in defect_ids
        or bool(set(as_ids(d.get("suites_ids"))) & suite_ids)
        or bool(set(as_ids(d.get("requirements_ids"))) & selected_req_ids)
    ]
    return suites, selected_reqs, selected_defects


def display_suite_details(bridge, suite):
    suite_id = str(suite.get("id"))
    try:
        details = bridge.get_suite_details(suite.get("id")) or {}
        if not isinstance(details, dict):
            details = {}
    except Exception as exc:
        st.warning(f"Détails indisponibles pour {suite_id} : {exc}")
        details = {}

    name = details.get("name") or suite.get("name") or "Suite sans nom"
    status = (
        details.get("last_result")
        or details.get("result")
        or details.get("status")
        or suite.get("last_result")
    )
    desc = details.get("description") or suite.get("description") or ""
    defects = as_ids(details.get("defects_ids", suite.get("defects_ids")))

    with st.container(border=True):
        st.markdown(f"#### {suite_id} · {name}")
        if desc:
            st.caption(str(desc))
        d1, d2, d3 = st.columns(3)
        d1.markdown(f"**Statut**  \n{status_badge(status)}")
        d2.markdown(f"**Code UID**  \n`{value_or_dash(details.get('code_uid', suite.get('code_uid')))}`")
        d3.markdown(f"**Faits techniques**  \n{len(defects)}")

        if defects:
            st.caption("Liens : " + ", ".join(defects))

        # Ces sous-onglets sont simples : ils n'ont aucune sélection à mémoriser.
        tab_steps, tab_logs, tab_history = st.tabs(["Étapes", "Console", "Historique"])

        with tab_steps:
            steps = records(details.get("steps"))
            if steps:
                df_steps = pd.DataFrame([
                    {
                        "ID": value_or_dash(s.get("id")),
                        "Étape": value_or_dash(s.get("name")),
                        "Statut": show_status(s.get("last_result", s.get("result"))),
                        "Sortie": value_or_dash(s.get("console_output")),
                    }
                    for s in steps
                ])
                as_table(df_steps, height=min(380, 64 + 36 * len(df_steps)))
            else:
                st.caption("Aucune étape enregistrée.")

        with tab_logs:
            st.code(str(details.get("console_output") or "Aucune sortie console."), language="text")

        with tab_history:
            history = records(details.get("history"))
            if history:
                df_hist = pd.DataFrame([
                    {
                        "Exécution": value_or_dash(h.get("run_id")),
                        "Date": value_or_dash(h.get("date")).replace("<span>:</span>", ":"),
                        "Statut": show_status(h.get("result", h.get("last_result"))),
                    }
                    for h in history
                ])
                as_table(df_hist, height=min(360, 64 + 36 * len(df_hist)))
            else:
                st.caption("Aucun historique disponible.")
